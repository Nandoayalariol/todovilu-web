        for (var i = 0; i < data.length; i++) {
          var row = data[i];
          if (!row[0] || String(row[0]).trim() === "") continue;
          var titulo = String(row[1]).toLowerCase(); 
          if (busqueda === "" || titulo.includes(busqueda)) {
             output.push(procesarFila(row, nombreHoja));

    }

  if (busqueda === "") { output.sort(function() { return 0.5 - Math.random() }); }

function procesarFila(row, categoria) {
  var meta = { link: "#", year: "----", generos: "General" };
  var datosOriginales = {}; // Aquí guardaremos el JSON completo para la App

  // INTENTO 1: Leer JSON (Columna E)
  try {
    var jsonRaw = row[4];
    if (jsonRaw && String(jsonRaw).trim() !== "") {
      var parsed = JSON.parse(jsonRaw);
      datosOriginales = parsed; // Guardamos el objeto original exacto

      if (parsed.year) meta.year = String(parsed.year);
      if (parsed.link) meta.link = parsed.link;
      if (parsed.generos) meta.generos = Array.isArray(parsed.generos) ? parsed.generos.join(", ") : parsed.generos;
      if (meta.link === "#" && parsed.trailerUrl) meta.link = parsed.trailerUrl;
      // Aseguramos que la App tenga su ID correcto
      if (parsed.idMsgCaratula) meta.idApp = parsed.idMsgCaratula;
    }
  } catch (e) {}

  // Si no había JSON, creamos uno básico para que la App no falle
  if (Object.keys(datosOriginales).length === 0) {
      datosOriginales = {
          id: String(row[0]),
          titulo: String(row[1]),
          imagen: String(row[2]),
          sinopsis: String(row[3]),
          link: row[5] || "#",
          categoria: categoria
      };
  }

  // INTENTO 2: Link Manual
  if (row[5] && String(row[5]).startsWith("http")) meta.link = row[5];

  return {
    id: String(row[0]),          
    idApp: String(meta.idApp || row[0]),   
    titulo: String(row[1]),      
    imagen: String(row[2]),
    sinopsis: String(row[3]),
    link: meta.link,
    anio: meta.year,
    generos: meta.generos,
    categoria: categoria,
    // 🔥 ESTA ES LA CLAVE: Enviamos el JSON crudo para las descargas
    jsonFull: datosOriginales 
  };
}

function servirArchivoJson(idBuscado, nombreHoja) {
  const data = buscarFila(idBuscado, nombreHoja);
  if (!data) return ContentService.createTextOutput("Error: No encontrado");

function servirPackJson(idsString, categoriaFija) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let jsonArray = [];
  let nombreArchivo = "pack_descarga.json";

  if (idsString === "TODO" && categoriaFija) {
    const sheet = ss.getSheetByName(categoriaFija);
    if (sheet && sheet.getLastRow() > 1) {
      const datos = sheet.getRange(2, 5, sheet.getLastRow() - 1, 1).getValues();
      jsonArray = datos.map(r => { try { return JSON.parse(r[0]); } catch(e) { return null; } }).filter(item => item !== null);
      nombreArchivo = "Pack_Completo_" + categoriaFija + ".json";
    }
  } else if (idsString) {
    const idsBuscados = idsString.split(",");
    const hojas = ['Peliculas', 'Series', 'Animes', 'Reelshorts', 'Textos'];
    hojas.forEach(nombre => {
      const sheet = ss.getSheetByName(nombre);
      if (sheet && sheet.getLastRow() > 1) {
        const datos = sheet.getRange(2, 1, sheet.getLastRow() - 1, 5).getValues();
        datos.forEach(fila => {
          if (idsBuscados.includes(String(fila[0]))) {
            try { jsonArray.push(JSON.parse(fila[4])); } catch(e) {}
          }
        });
      }
    });
    nombreArchivo = "Pack_Seleccion_" + jsonArray.length + "_items.json";
  }
  return ContentService.createTextOutput(JSON.stringify(jsonArray)).setMimeType(ContentService.MimeType.JSON).downloadAsFile(nombreArchivo);
}

function buscarFila(id, hoja) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(hoja);
  if (!sheet) return null;
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === String(id)) return data[i];
  }
  return null;
}

function limpiarNombre(txt) {
  let s = txt.normalize("NFD").replace(/[\u0300-\u036f]/g, "");
  s = s.replace(/ñ/gi, "n").replace(/[^a-zA-Z0-9 \-]/g, "");
  return s.trim();
}

function normalizarTexto(texto) { return String(texto).normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase().trim(); }

function buscarCandidatos(query) {
  try {
    const url = `https://api.themoviedb.org/3/search/multi?api_key=${API_KEY}&language=es-MX&query=${encodeURIComponent(query)}`;
    const data = JSON.parse(UrlFetchApp.fetch(url, {muteHttpExceptions:true}).getContentText());
    if (!data.results) return [];
    return data.results.map(item => ({
      id: item.id, tipo: item.media_type, titulo: item.title || item.name,
      anio: item.release_date || item.first_air_date || '?',
      imagen: item.poster_path ? `https://image.tmdb.org/t/p/w92${item.poster_path}` : 'https://via.placeholder.com/92x138?text=Sin+IMG',
      descripcion: item.overview ? item.overview.substring(0, 50) + '...' : ''
    }));
  } catch (e) { return []; }
}

function buscarEnBaseDeDatos(query) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojas = ['Peliculas', 'Series', 'Animes', 'Reelshorts', 'Textos'];
  let resultados = [];
  const qClean = normalizarTexto(query);
  hojas.forEach(nombre => {
    const sheet = ss.getSheetByName(nombre);
    if (sheet && sheet.getLastRow() > 1) {
      const datos = sheet.getRange(2, 1, sheet.getLastRow() - 1, 5).getValues();
      datos.forEach(fila => {
        const id = normalizarTexto(fila[0]);
        const titulo = normalizarTexto(fila[1]);
        if (titulo.includes(qClean) || id === qClean) {
          resultados.push({ json: fila[4], titulo: fila[1], imagen: fila[2], hoja: nombre });
        }
      });
    }
  });
  return resultados;
}

function obtenerDatosPeli(id, tipo) {
  let append = "credits,videos,release_dates,content_ratings";
  if (tipo === 'tv') append = "credits,videos,content_ratings";
  const url = `https://api.themoviedb.org/3/${tipo}/${id}?api_key=${API_KEY}&language=es-MX&append_to_response=${append}`;
  const item = JSON.parse(UrlFetchApp.fetch(url).getContentText());
  
  let duracion = 0;
  if (tipo === 'movie') duracion = item.runtime || 0;
  else if (item.episode_run_time?.length) duracion = item.episode_run_time[0];

  const trailer = buscarTrailer(item.videos);
  const clasif = buscarClasificacion(item, tipo);
  const director = buscarDirector(item.credits);
  const reparto = (item.credits && item.credits.cast) ? item.credits.cast.slice(0, 8).map(c => c.name) : [];
  const productoras = (item.production_companies) ? item.production_companies.map(p => p.name) : [];
  const creadores = (item.created_by) ? item.created_by.map(c => c.name) : [];
  const networks = (item.networks) ? item.networks.map(n => n.name) : [];
  
  // 🔥 AUTO-DETECTAR SAGA / COLECCIÓN
  let sagaAuto = "";
  if (item.belongs_to_collection && item.belongs_to_collection.name) {
      // Convertir "The Avengers Collection" -> "the_avengers_collection"
      sagaAuto = item.belongs_to_collection.name.toLowerCase().trim().replace(/ /g, "_").replace(/[^a-z0-9_]/g, "");
  }

  let datos = {
    tmdbId: item.id, titulo: item.title || item.name,
    original_name: item.original_name || item.original_title,
    sinopsis: item.overview,
    imagen: item.poster_path ? `https://image.tmdb.org/t/p/w500${item.poster_path}` : "",
    year: parseInt((item.release_date || item.first_air_date || "").substring(0,4)) || 0,
    duracion: duracion, generos: item.genres.map(g => g.name),
    tipo: tipo, categoria_sugerida: (tipo === 'movie') ? 'peliculas' : 'series',
    calificacion: item.vote_average || 0.0, clasificacion: clasif,
    idioma: item.original_language || "es", paisOrigen: (item.production_countries && item.production_countries.length > 0) ? item.production_countries[0].iso_3166_1 : "",
    reparto: reparto, productoras: productoras, trailerUrl: trailer,
    director: director, creadores: creadores, networks: networks,
    enEmision: (item.status === "Returning Series" || item.status === "In Production"),
    saga: sagaAuto, // <--- CAMPO AUTOMÁTICO
    episodios: [] 
  };

  if (tipo !== 'movie') {
    datos.episodios = traerEpisodios(id, item.number_of_seasons);
    datos.total_temporadas = item.number_of_seasons || 0;
  }
  return datos;
}

function buscarTrailer(videosObj) {
  if (!videosObj || !videosObj.results) return "";
  const vid = videosObj.results.find(v => v.type === "Trailer" && v.site === "YouTube");
  if (vid) return `https://www.youtube.com/watch?v=${vid.key}`;
  return "";
}

function buscarClasificacion(item, tipo) {
  try {
    if (tipo === 'movie' && item.release_dates) {
      const release = item.release_dates.results.find(r => r.iso_3166_1 === "MX") || item.release_dates.results.find(r => r.iso_3166_1 === "US");
      if (release && release.release_dates.length > 0) return release.release_dates[0].certification;
    } else if (tipo === 'tv' && item.content_ratings) {
      const rating = item.content_ratings.results.find(r => r.iso_3166_1 === "MX") || item.content_ratings.results.find(r => r.iso_3166_1 === "US");
      if (rating) return rating.rating;
    }
  } catch(e) {}
  return "";
}

function buscarDirector(credits) {
  if (!credits || !credits.crew) return "";
  const dir = credits.crew.find(c => c.job === "Director");
  return dir ? dir.name : "";
}

function traerEpisodios(id, totalTemps) {
  let lista = [];
  for(let s=1; s<=Math.min(totalTemps, 20); s++){
     try {
       let urlS = `https://api.themoviedb.org/3/tv/${id}/season/${s}?api_key=${API_KEY}&language=es-MX`;
       let data = JSON.parse(UrlFetchApp.fetch(urlS).getContentText());
       data.episodes.forEach(e => {
         lista.push({ temporada: s, numero: e.episode_number, titulo: e.name, sinopsis: e.overview || "", duracion: e.runtime || 0 });
       });
     } catch(e) {}
  }
  return lista;
}

function upsertFirestore(firestore, ruta, datos) {
  try { firestore.updateDocument(ruta, datos); } catch(e) {
    try { firestore.createDocument(ruta, datos); } catch(e2) {
      try { firestore.deleteDocument(ruta); } catch(e3){}
      firestore.createDocument(ruta, datos);
    }
  }
}

function guardarFormulario(d) {

  const nombreHoja = catFinal.charAt(0).toUpperCase() + catFinal.slice(1);

  if(!sheet) return "Error: No existe pestaña " + nombreHoja;

  const data = sheet.getDataRange().getValues();
  let filaEncontrada = -1;

  for (let i = 1; i < data.length; i++) {
    let idEnCelda = String(data[i][0]).trim();
    if (idEnCelda === tmdbKey) {
      filaEncontrada = i + 1;
      try { 
        let jsonViejo = JSON.parse(data[i][4]); 
        if (jsonViejo.fechaAgregada) fechaOriginal = jsonViejo.fechaAgregada;
      } catch(e){}
      break;
    }
  }

  const fechaApp = fechaOriginal || { "_seconds": Math.floor(ahora.getTime() / 1000), "_nanoseconds": 0 };
  let fechaFirestore = ahora;
  if (fechaOriginal && fechaOriginal._seconds) {
      fechaFirestore = new Date(fechaOriginal._seconds * 1000);
  }

  // --- DATOS BASE ---
  let baseData = {
    titulo: d.titulo, sinopsis: d.sinopsis, year: parseInt(d.anio) || 0,
    destacada: d.destacada === true, nuevo: d.nuevo === true,
    generos: d.generos, tmdbId: parseInt(d.tmdbId),
    categoria: catFinal, idCanalCaratula: idCanalFijo, idMsgCaratula: finalMsgCaratula,
    calificacion: parseFloat(d.calificacion) || 0, clasificacion: d.clasificacion || "",
    idioma: d.idioma || "es", paisOrigen: d.paisOrigen || "", trailerUrl: d.trailerUrl || "",
    reparto: d.reparto || [], productoras: d.productoras || [],
    saga: d.saga || "", // <--- GUARDAMOS SAGA DETECTADA O MANUAL
    fechaAgregada: fechaApp
  };

  let jsonFirestore = { ...baseData, fechaAgregada: fechaFirestore, lastUpdate: ahora };
  let jsonExcel = { ...baseData, fechaAgregada: fechaApp, lastUpdate: fechaApp }; 

  const rutaDoc = catFinal + "/" + finalMsgCaratula;
  const firestore = getFirestore();

  try {
    if (catFinal === 'peliculas' || catFinal === 'reelshorts') {
      let rawMsgVideo = parseInt(d.idMsgVideo) || 0;
      [jsonFirestore, jsonExcel].forEach(j => {
        j.idCanalVideo = idCanalFijo;
        j.idMsgVideo = rawMsgVideo > 0 ? rawMsgVideo * MULTIPLIER : 0;
        j.duracion = parseInt(d.duracion) || 0;
        j.director = d.director || "";
      });
      upsertFirestore(firestore, rutaDoc, jsonFirestore);
    } 
    else if (catFinal === 'series' || catFinal === 'animes') {
      [jsonFirestore, jsonExcel].forEach(j => {
        j.temporadas = d.total_temporadas;
        j.duracionPromedio = parseInt(d.duracion) || 0;
        j.creadores = d.creadores || [];
        j.networks = d.networks || [];
        j.enEmision = d.enEmision === true;
        if (catFinal === 'animes') j.estado = "En emisión"; 
      });

      let episodiosListos = d.episodios.map(ep => {
        let finalEpMsg = (parseInt(ep.idMsgVideo) || 0) * MULTIPLIER;
        return {
          numero: ep.numero, titulo: ep.titulo, sinopsis: ep.sinopsis, duracion: ep.duracion || 0, temporada: ep.temporada,
          idCanalVideo: idCanalFijo, idMsgVideo: finalEpMsg, idCanalCaratula: idCanalFijo, idMsgCaratula: 0,
          subirAFirestore: ep.subirAFirestore
        };
      });
      jsonExcel.episodios = episodiosListos;
      upsertFirestore(firestore, rutaDoc, jsonFirestore); 
      episodiosListos.forEach(ep => {
        if (ep.subirAFirestore === true) {
            delete ep.subirAFirestore; 
            let nombreDoc = (ep.temporada > 1) ? "s"+ep.temporada+"ep"+ep.numero : "ep"+ep.numero;
            let rutaEp = rutaDoc + "/episodios/" + nombreDoc;
            upsertFirestore(firestore, rutaEp, ep);
        }
      });
    }
    else if (catFinal === 'textos') {

      });
      let capitulosListos = d.episodios.map(ep => {

      });
      jsonExcel.capitulos = capitulosListos;
      upsertFirestore(firestore, rutaDoc, jsonFirestore);
    }
  } catch(e) { return "⚠️ Error Firestore: " + e.message; }

  if (filaEncontrada > 0) {
    sheet.getRange(filaEncontrada, 2).setValue(d.titulo);
    sheet.getRange(filaEncontrada, 3).setValue(d.imagen);
    sheet.getRange(filaEncontrada, 4).setValue(d.sinopsis);
    sheet.getRange(filaEncontrada, 5).setValue(JSON.stringify(jsonExcel));
    sheet.getRange(filaEncontrada, 6).setValue(new Date()); 
    return "✅ [EDITADO] " + d.titulo;
  } else {
    sheet.appendRow([d.tmdbId, d.titulo, d.imagen, d.sinopsis, JSON.stringify(jsonExcel), new Date()]);
    return "✅ [NUEVO] " + d.titulo;
  }
}

function obtenerTodoElCatalogo() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojas = ['Peliculas', 'Series', 'Animes', 'Reelshorts', 'Textos'];
  let todo = [];
  hojas.forEach(nombre => {
    const sheet = ss.getSheetByName(nombre);
    if (sheet && sheet.getLastRow() > 1) {
      const datos = sheet.getRange(2, 1, sheet.getLastRow() - 1, 5).getValues();
      const datosConTipo = datos.map(f => [ String(f[0]), String(f[1]), String(f[2]), String(f[3]), String(f[4]), nombre ]);
      todo = todo.concat(datosConTipo);
    }
  });
  for (let i = todo.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [todo[i], todo[j]] = [todo[j], todo[i]]; }
  return todo;
}

function obtenerDatosHoja(n) { 
  const s = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(n); 
  if (!s || s.getLastRow() < 2) return []; 
  return s.getRange(2, 1, s.getLastRow()-1, 5).getValues().map(r=>r.map(String)); 
}
